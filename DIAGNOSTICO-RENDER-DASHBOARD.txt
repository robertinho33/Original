async function loadModuleData(section) {

    const content =
        root.querySelector(
            ".admin-loading"
        );

    if (!content) {
        return;
    }


    try {

        let response;


        switch (section) {

            case "orders":
                response =
                    await adminApi.orders();
                break;

            case "products":
                response =
                    await adminApi.products();
                break;

            case "categories":
                response =
                    await adminApi.categories();
                break;

            case "inventory":
                response =
                    await adminApi.inventory();
                break;

            case "customers":
                response =
                    await adminApi.customers();
                break;

            case "finance":
                response =

                    await adminApi.coupons();
                break;

            case "logistics":
                response =
                    await adminApi.logistics();
                break;

            case "reports":
                response =
                    await adminApi.reports();
                break;

            case "audit":
                response =
                    await adminApi.audit();
                break;

            case "settings":
                response =
                    await adminApi.settings();
                break;

            default:
                response = {
                    success: true,
                    data: []
                };
        }


        const data =
            response?.data ?? [];


        if (section === "orders") {

            renderOrders(
                content,
                data
            );

            return;
        }


        if (section === "products") {

            renderProducts(

                data
            );

            return;
        }


        if (section === "categories") {

            renderCategories(
                content,
                data
            );

            return;
        }


        if (section === "inventory") {

            renderInventory(
                content,
                data
            );

            return;
        }


        if (section === "customers") {

            renderCustomers(
                content,
                data
            );

            return;

        if (section === "finance") {

            renderFinance(

            return;

            renderLogistics(
                content,
                data
            );

            return;
        }


        if (section === "reports") {

            renderReports(
                content,
                data
            );

            return;
        }


        if (section === "audit") {

            renderAudit(
                content,
                data
            );

            return;
